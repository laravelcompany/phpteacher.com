class UrlToArticleRef extends TamperBase implements ContainerFactoryPluginInterface {

  /**
   * Path Alias.
   *
   * @var Drupal\path_alias\AliasManagerInterface
   */
  protected AliasManagerInterface $pathAliasManager;

  /**
   * Constructs a new URL To Article Ref Tamper object.
   *
   * @inheritDoc
   */
  public function __construct(array $configuration, $plugin_id, $plugin_definition, SourceDefinitionInterface $source_definition, AliasManagerInterface $path_alias_manager) {
    parent::__construct($configuration, $plugin_id, $plugin_definition, $source_definition);
    $this->pathAliasManager = $path_alias_manager;
  }

  /**
   * Dependency injection for Article Ref tamper object.
   *
   * @inheritDoc
   */
  public static function create(ContainerInterface $container, array $configuration, $plugin_id, $plugin_definition) {
    return new static(
      $configuration,
      $plugin_id,
      $plugin_definition,
      $configuration['source_definition'],
      $container->get('path_alias.manager')
    );
  }
