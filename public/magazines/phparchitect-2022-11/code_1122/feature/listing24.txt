class Color
{
    // ...
    public function isEqual(Color $otherColor): bool
    {
        return $this->red === $otherColor->red
            && $this->green === $otherColor->green
            && $this->blue === $otherColor->blue
            && $this->alpha === $otherColor->alpha;
    }
}
