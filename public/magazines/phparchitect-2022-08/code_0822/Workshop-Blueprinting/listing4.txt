<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class GameTag extends Model
{
    use HasFactory;

    protected $fillable = [
        'name',
        'game_id',
    ];

    protected $casts = [
        'id' => 'integer',
        'game_id' => 'integer',
    ];

    public function game()
    {
        return $this->belongsTo(Game::class);
    }
}
