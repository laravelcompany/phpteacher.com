!theme metal
skinparam defaultTextAlignment center

' Construction of the objects

Users(visitors, "Visitors", "")

cloud AWS {
  APIGateway(api, "Upload API", "")
  Lambda(lambda, "Converter\nLambda", "")
  DynamoDB(storyDb, "Story Data\nDynamoDB", "")
  EC2(ec2, "Extractor\nEC2", "")

  rectangle {
    SimpleStorageService(s3, "S3", "")
    folder thumbnails
    folder videos
  }
}

interface "https" as apiEndpoint
interface "https" as cdnEndpoint

' Bind interfaces to objets

apiEndpoint . api
cdnEndpoint . thumbnails

' Create the arrows

visitors -> apiEndpoint : "Upload\na Story"
api-> lambda: "Trigger\ncropping / conversion"
lambda --> videos: storage
videos --> ec2
ec2 -> storyDb: "insert\nmetadata"
ec2 -> thumbnails: storage
visitors <-> cdnEndpoint: "browse\nthe images"
