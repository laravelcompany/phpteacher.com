<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Game extends Model
{
    use HasFactory;

    protected $fillable = [
        'name',
        'slug',
        'site_url',
        'publisher_url',
        'publisher',
        'description',
        'thumbnail_url',
        'screen_url',
        'published',
    ];

    protected $casts = [
        'id' => 'integer',
        'published' => 'boolean',
    ];

    public function gameTags()
    {
        return $this->hasMany(GameTag::class);
    }
}
